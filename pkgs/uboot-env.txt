# The stored U-Boot environment (p7 `env`), DELTA ONLY.
#
# Everything the bootloader itself needs -- bootcmd, preboot, altbootcmd, the
# load addresses, bootpart, the milestone words -- is COMPILED INTO mainline
# U-Boot and is lifted from the built image at pack time (pkgs/uboot-env.nix).
# Only variables that do not exist in the compiled-in default belong here:
# state somebody writes at runtime, and which has to survive a reboot.
#
# bootcount / upgrade_available ARE NOT HERE ANY MORE (#89 rung 5).
#   The counter moved OUT of the environment and into
#   TOP_CHIPMODE_GLB_BACKUP1 (0x02390030) behind DM_BOOTCOUNT_SYSCON, because
#   patch 0020 stopped U-Boot reading this partition at all and
#   CONFIG_BOOTCOUNT_ENV needs a writable environment. A stored `bootcount`
#   here would be a number nothing reads and nothing updates -- worse than
#   absent, because `fw_printenv bootcount` would answer it. `upgrade_available`
#   went with it: it was the CONFIG_BOOTCOUNT_ENV gate that kept U-Boot from
#   writing the environment on every boot, and a register write needs no gate.
#   The counter is read and cleared with `devmem 0x02390030 32`; the appliance's
#   nanokvm-mark-good unit does it once a boot has proven healthy.
#
# bootsystem
#   Read by `nanokvm-checkboot.service`, which re-arms the A/B slot register
#   to the slot named here. Under the VENDOR U-Boot this was an output --
#   `set_slot_ab()` read the register and wrote it back every boot. Mainline
#   U-Boot has no A/B logic at all, and since #97 there are no `_b` twins to
#   select between, so it is a constant: "A". Keeping it constant is what makes
#   the milestone register a readable oracle instead of a value that alternates
#   every boot. The rollback is `bootcount` and two extlinux configs, not the
#   slot bits.
#
# WHAT IS DELIBERATELY ABSENT. `bootargs`: mainline U-Boot's bootcmd runs
# `sysboot`, which builds the kernel command line from the APPEND line of
# /boot/extlinux/extlinux.conf and overwrites `bootargs` with it. A stored
# bootargs would be read by nothing.
bootsystem=A
